## Usage

```tsx
import { ScrollFog } from "@seed-design/react";
```

```tsx
<div style={{ maxHeight: "300px" }}>
  <ScrollFog>
    {/* 스크롤 가능한 콘텐츠 */}
  </ScrollFog>
</div>
```

## Anatomy

```tsx
<div> {/* Parent가 크기 제어 */}
  <ScrollFog
    placement={["top", "bottom"]}
    size={20}
    hideScrollBar={false}
  >
    {children}
  </ScrollFog>
</div>
```
